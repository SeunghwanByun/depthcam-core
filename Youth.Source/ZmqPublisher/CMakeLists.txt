find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(ZMQ libzmq)
endif()

if(NOT ZMQ_FOUND)
    find_library(ZMQ_LIBRARIES NAMES zmq)
    find_path(ZMQ_INCLUDE_DIRS NAMES zmq.h)
    if(ZMQ_LIBRARIES AND ZMQ_INCLUDE_DIRS)
        set(ZMQ_FOUND TRUE)
    endif()
endif()

if(ZMQ_FOUND)
    add_library(ZmqPublisherLib STATIC zmq_publisher.c)
    target_include_directories(ZmqPublisherLib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${ZMQ_INCLUDE_DIRS})
    target_link_libraries(ZmqPublisherLib PUBLIC ${ZMQ_LIBRARIES})
    target_compile_definitions(ZmqPublisherLib PUBLIC YOUTH_ZMQ_ENABLED)
    message(STATUS "[ZmqPublisher] ZMQ found — depth frame publishing enabled")
else()
    # Stub library so the rest of the build doesn't break without ZMQ
    add_library(ZmqPublisherLib INTERFACE)
    target_include_directories(ZmqPublisherLib INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
    message(STATUS "[ZmqPublisher] ZMQ not found — using stub (no publishing)")
endif()
